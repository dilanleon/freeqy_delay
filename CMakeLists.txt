cmake_minimum_required(VERSION 3.22)
project(SpectralDelay VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
option(SPECTRAL_DIAGNOSTIC "Input FFT and diagnostic sine output" OFF)
if(JUCE_DIR)
 add_subdirectory(${JUCE_DIR} JUCE)
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG d60b7a78030661574a615b586d9325ba7a7fa7dd GIT_SHALLOW FALSE)
 FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(SpectralDelay COMPANY_NAME "DLeon" BUNDLE_ID "com.dleon.spectraldelay" PLUGIN_MANUFACTURER_CODE Dleo PLUGIN_CODE Spdl FORMATS VST3 Standalone PRODUCT_NAME "Spectral Delay" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE MICROPHONE_PERMISSION_ENABLED TRUE MICROPHONE_PERMISSION_TEXT "Analyze and process audio input")
target_sources(SpectralDelay PRIVATE Source/Engine.cpp Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(SpectralDelay PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 SPECTRAL_DIAGNOSTIC=$<BOOL:${SPECTRAL_DIAGNOSTIC}>)
target_link_libraries(SpectralDelay PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
juce_add_console_app(DspTests PRODUCT_NAME "DSP Tests")
target_sources(DspTests PRIVATE Tests/Main.cpp Source/Engine.cpp)
target_include_directories(DspTests PRIVATE Source)
target_compile_definitions(DspTests PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(DspTests PRIVATE juce::juce_dsp)
add_test(NAME dsp COMMAND DspTests)
juce_add_console_app(HostCheck PRODUCT_NAME "VST3 Host Check")
target_sources(HostCheck PRIVATE Tests/HostCheck.cpp)
add_dependencies(HostCheck SpectralDelay_VST3)
target_compile_definitions(HostCheck PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(HostCheck PRIVATE juce::juce_audio_processors)
if(SPECTRAL_DIAGNOSTIC)
 add_test(NAME vst3_host COMMAND HostCheck "$<TARGET_BUNDLE_DIR:SpectralDelay_VST3>" --diagnostic)
else()
 add_test(NAME vst3_host COMMAND HostCheck "$<TARGET_BUNDLE_DIR:SpectralDelay_VST3>")
endif()
add_executable(StateTests)
target_sources(StateTests PRIVATE Tests/StateTests.cpp)
target_include_directories(StateTests PRIVATE Source)
target_link_libraries(StateTests PRIVATE SpectralDelay juce::juce_audio_utils juce::juce_dsp)
add_test(NAME bank_state COMMAND StateTests)
set_tests_properties(vst3_host bank_state PROPERTIES TIMEOUT 30)
set_tests_properties(dsp PROPERTIES TIMEOUT 120)
